```verilog
//======================================================
// 1x8 DEMULTIPLEXER
// 1 Data Input, 3 Select Lines, 8 Outputs
//======================================================

module demux_1x8 (
    input D,
    input S2,
    input S1,
    input S0,
    output reg Y0,
    output reg Y1,
    output reg Y2,
    output reg Y3,
    output reg Y4,
    output reg Y5,
    output reg Y6,
    output reg Y7
);

    always @(*) begin

        // Default outputs
        Y0 = 1'b0;
        Y1 = 1'b0;
        Y2 = 1'b0;
        Y3 = 1'b0;
        Y4 = 1'b0;
        Y5 = 1'b0;
        Y6 = 1'b0;
        Y7 = 1'b0;

        case ({S2, S1, S0})

            3'b000: Y0 = D;
            3'b001: Y1 = D;
            3'b010: Y2 = D;
            3'b011: Y3 = D;
            3'b100: Y4 = D;
            3'b101: Y5 = D;
            3'b110: Y6 = D;
            3'b111: Y7 = D;

        endcase
    end

endmodule
```
